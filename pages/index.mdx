import 'package:flutter/material.dart';
import 'package:http/http.dart' as http;
import 'dart:convert';

void main() {
  runApp(AngelicoDiorApp());
}

class AngelicoDiorApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Angelico Dior AI',
      theme: ThemeData(
        brightness: Brightness.dark,
        scaffoldBackgroundColor: Colors.black,
        textTheme: TextTheme(
          bodyMedium: TextStyle(color: Colors.white),
        ),
      ),
      home: ChatPage(),
    );
  }
}

class ChatPage extends StatefulWidget {
  @override
  _ChatPageState createState() => _ChatPageState();
}

class _ChatPageState extends State<ChatPage> {
  TextEditingController controller = TextEditingController();
  List<Map<String, String>> messages = [];

  Future<void> sendMessage() async {
    final userText = controller.text.trim();

    if (userText.isEmpty) return;

    setState(() {
      messages.add({"role": "user", "text": userText});
    });

    controller.clear();

    try {
      final response = await http.post(
        Uri.parse("http://SEU_BACKEND_AQUI/chat"),
        headers: {"Content-Type": "application/json"},
        body: jsonEncode({"text": userText}),
      );

      final data = jsonDecode(response.body);

      setState(() {
        messages.add({"role": "assistant", "text": data["reply"]});
      });
    } catch (e) {
      setState(() {
        messages.add({"role": "assistant", "text": "⚠ Erro ao conectar ao servidor."});
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text("Angelico Dior"),
        centerTitle: true,
        backgroundColor: Colors.black,
      ),
      body: Column(
        children: [
          Expanded(
            child: ListView(
              children: messages.map((msg) {
                return Container(
                  padding: EdgeInsets.all(12),
                  margin: EdgeInsets.all(8),
                  alignment: msg["role"] == "user"
                      ? Alignment.centerRight
                      : Alignment.centerLeft,
                  decoration: BoxDecoration(
                    color: msg["role"] == "user"
                        ? Colors.blueGrey
                        : Colors.grey[900],
                    borderRadius: BorderRadius.circular(10),
                  ),
                  child: Text(msg["text"] ?? ""),
                );
              }).toList(),
            ),
          ),
          Padding(
            padding: EdgeInsets.all(10),
            child: Row(
              children: [
                Expanded(
                  child: TextField(
                    controller: controller,
                    style: TextStyle(color: Colors.white),
                    decoration: InputDecoration(
                      hintText: "Digite sua mensagem...",
                      hintStyle: TextStyle(color: Colors.grey),
                      enabledBorder: UnderlineInputBorder(
                        borderSide: BorderSide(color: Colors.white38),
                      ),
                    ),
                  ),
                ),
                IconButton(
                  icon: Icon(Icons.send, color: Colors.white),
                  onPressed: sendMessage,
                )
              ],
            ),
          )
        ],
      ),
    );
  }
}
